从SFINAE到Concepts

 2026.7.31  2026.7.31  1387  7 mins

了解到C++20新出的Concepts解决了原本SFINAE的问题，还是蛮惊喜的，这东西好久没遇到过了，现在梳理一下吧。

SFINAE是C++模板元编程中的一个重要概念，全称是 “Substitution Failure Is Not An Error”（替换失败不是错误）。

那么为什么我们需要SFINAE呢，他要解决的是模板的重载问题，正式说法是"如何在编译期根据类型特性，有选择地启用/禁用模板"。类比函数重载来说，函数重载

的规则是：编译器根据参数类型直接匹配，这是编译器的"硬编码"规则。

void process(int x) {
    std::cout << "int: " << x << std::endl;
}

void process(double x) {
    std::cout << "double: " << x << std::endl;
}

void process(const std::string& x) {
    std::cout << "string: " << x << std::endl;
}

int main() {
    process(42);        // 调用 int 版本
    process(3.14);      // 调用 double 版本
    process("hello");    // 调用 string 版本
}

但模板是在编译期的东西，编译期才不知道你的template T是否具有相关操作

#include <iostream>

template<class T> auto print(const T& t, int) -> decltype(void(t.value))
{
    std::cout << "Has value: " << t.value << "\n";
}

template<class T> void print(const T& t, ...)
{
    std::cout << "General: " << t << "\n";
}

struct Foo { int value = 42; };

int main() {
    print(Foo{}, 0);   // Has value: 42
    print(3.14, 0);   // General: 3.14
}

这是一个经典的SFINAE实现，print(b)在这里因为SFINAE匹配特化版本失败，去匹配通用版本成功了(模板会优先匹配更高特化的版本，没有SFINAE的保护这里就会编译错误)。

decltype(void(t.value))是一种典型的SFINAE，若t.value不存在就换一个模板。

除此之外还有很多方法触发SFINAE，只要能让模板替换失败就可以，比如函数返回值，函数参数，模板参数…反正有很多

在这里讲一下enable_if算常用点的

template<typename T,
         typename std::enable_if<std::is_integral<T>::value, void>::type* = nullptr>

enable_if<std::is_integral::value当T是整形value为true，enable_if<>::type*为void类型指针，不是整形SFINAE

以上晦涩难懂且编译信息复杂错误信息难以定位，于是我们引入Concepts

C++ Concepts（概念） 是 C++20 引入的一套模板约束机制。它可以明确描述

“这个模板参数必须具备哪些能力，才能使用该模板。”

它主要解决传统模板中错误信息难懂、接口要求不明确、SFINAE 写法复杂等问题。

以下为Concepts参考


// concepts_demo.cpp
// 编译：
//   g++ -std=c++20 -Wall -Wextra -pedantic concepts_demo.cpp -o concepts_demo
//   clang++ -std=c++20 -Wall -Wextra -pedantic concepts_demo.cpp -o concepts_demo

#include <concepts>
#include <cstddef>
#include <functional>
#include <iostream>
#include <list>
#include <ostream>
#include <string>
#include <type_traits>
#include <utility>
#include <vector>

// ============================================================
// 1. 使用 type_traits 定义 Concept
// ============================================================

template<typename T>
concept Integer = std::is_integral_v<T>;

template<typename T>
concept Number =
std::integral<T> ||
std::floating_point<T>;

// 通过组合 Concept 建立更具体的约束。
// 这样有利于编译器进行约束包含关系（subsumption）判断。
template<typename T>
concept SignedInteger =
std::integral<T> &&
std::signed_integral<T>;

// ============================================================
// 2. 使用 requires 表达式定义能力型 Concept
// ============================================================

// 要求 T 支持 a + b，并且结果可以转换为 T。
template<typename T>
concept Addable = requires(const T & a, const T & b)
{
    { a + b } -> std::convertible_to<T>;
};

// 要求 T 可以输出到 std::ostream。
template<typename T>
concept StreamPrintable =
    requires(std::ostream & output, const T & value)
{
    { output << value } -> std::same_as<std::ostream&>;
};

// 要求 T 存在 value_type。
template<typename T>
concept HasValueType = requires
{
    typename T::value_type;
};

// ============================================================
// 3. requires 表达式中的四种要求
// ============================================================

template<typename T>
concept ContainerLike = requires(T container, const T constContainer)
{
    // 1）类型要求：
    // 检查嵌套类型是否存在。
    typename T::value_type;
    typename T::iterator;
    typename T::const_iterator;

    // 2）简单要求：
    // 只检查表达式是否合法。
    container.begin();
    container.end();
    constContainer.begin();
    constContainer.end();

    // 3）复合要求：
    // 检查表达式是否合法，并检查返回值约束。
    { constContainer.size() } -> std::convertible_to<std::size_t>;

    // 4）嵌套要求：
    // 在 requires 表达式内部增加布尔约束。
        requires std::copy_constructible<typename T::value_type>;
};

// 要求成员 swap 是 noexcept 的。
template<typename T>
concept NoThrowMemberSwappable =
    requires(T a, T b)
{
    { a.swap(b) } noexcept;
};

// ============================================================
// 4. 使用 Concept 的四种经典语法
// ============================================================

// 写法一：requires 子句。
template<typename T>
    requires Integer<T>
T twice_v1(T value)
{
    return value * 2;
}

// 写法二：直接约束模板参数。
template<Integer T>
T twice_v2(T value)
{
    return value * 2;
}

// 写法三：尾部 requires 子句。
template<typename T>
T twice_v3(T value) requires Integer<T>
{
    return value * 2;
}

// 写法四：简写函数模板，使用 constrained auto。
auto twice_v4(Integer auto value)
{
    return value * 2;
}

// 也可以直接使用标准库 Concept。
auto square(std::integral auto value)
{
    return value * value;
}

// ============================================================
// 5. 自定义 Concept 用于普通函数模板
// ============================================================

template<Addable T>
T add(const T& left, const T& right)
{
    return left + right;
}

template<StreamPrintable T>
void printLine(const T& value)
{
    std::cout << value << '\n';
}

// ============================================================
// 6. 多个 Concept 组合
// ============================================================

template<typename T>
concept PrintableNumber =
Number<T> &&
StreamPrintable<T>;

template<PrintableNumber T>
void printNumber(T value)
{
    std::cout << "number = " << value << '\n';
}

// ============================================================
// 7. Concepts 参与函数重载
// ============================================================

// 无约束的通用版本。
template<typename T>
void classify(const T&)
{
    std::cout << "generic type\n";
}

// 整数版本比通用版本约束更强。
template<std::integral T>
void classify(const T&)
{
    std::cout << "integral type\n";
}

// SignedInteger 显式组合了 std::integral，
// 因而对有符号整数来说，它比 integral 版本更具体。
template<SignedInteger T>
void classify(const T&)
{
    std::cout << "signed integral type\n";
}

// ============================================================
// 8. Concept 约束类模板
// ============================================================

template<std::integral T>
class Counter
{
public:
    void add(T value)
    {
        count_ += value;
    }

    [[nodiscard]]
    T value() const
    {
        return count_;
    }

private:
    T count_{};
};

// ============================================================
// 9. 只约束类模板的部分成员函数
// ============================================================

template<typename T>
class Box
{
public:
    explicit Box(T value)
        : value_(std::move(value))
    {
    }

    [[nodiscard]]
    const T& value() const
    {
        return value_;
    }

    // 只有支持加法的类型才拥有 add() 能力。
    T add(const T& other) const
        requires Addable<T>
    {
        return value_ + other;
    }

    // 只有可输出类型才能调用 print()。
    void print() const
        requires StreamPrintable<T>
    {
        std::cout << "Box value: " << value_ << '\n';
    }

private:
    T value_;
};

// ============================================================
// 10. Range / Container Concept
// ============================================================

template<ContainerLike Container>
void printContainer(const Container& container)
{
    std::cout << "[ ";

    for (const auto& element : container)
    {
        std::cout << element << ' ';
    }

    std::cout << "]\n";
}

// 使用 C++20 Ranges 自带的 Concept。
template<std::ranges::input_range Range>
    requires StreamPrintable<std::ranges::range_value_t<Range>>
void printRange(const Range& range)
{
    std::cout << "range: ";

    for (const auto& element : range)
    {
        std::cout << element << ' ';
    }

    std::cout << '\n';
}

// ============================================================
// 11. 可调用对象 Concept
// ============================================================

template<typename Function, typename... Args>
    requires std::invocable<Function, Args...>
decltype(auto) execute(Function&& function, Args&&... args)
{
    return std::invoke(
        std::forward<Function>(function),
        std::forward<Args>(args)...
    );
}

// 对返回值进一步施加约束。
template<typename Function, typename... Args>
    requires std::regular_invocable<Function, Args...>
&& std::integral<
    std::invoke_result_t<Function, Args...>
>
auto executeIntegerFunction(Function&& function, Args&&... args)
{
    return std::invoke(
        std::forward<Function>(function),
        std::forward<Args>(args)...
    );
}

// ============================================================
// 12. 自定义类型满足 Concept
// ============================================================

struct User
{
    std::string name;
};

// 让 User 满足 StreamPrintable。
std::ostream& operator<<(std::ostream& output, const User& user)
{
    return output << "User{name=\"" << user.name << "\"}";
}

struct Point
{
    int x{};
    int y{};
};

// 让 Point 满足 Addable。
Point operator+(const Point& left, const Point& right)
{
    return {
        left.x + right.x,
        left.y + right.y
    };
}

// 让 Point 满足 StreamPrintable。
std::ostream& operator<<(std::ostream& output, const Point& point)
{
    return output << "Point{" << point.x << ", " << point.y << '}';
}

// 不满足 Addable，也不满足 StreamPrintable。
struct EmptyType
{
};

// ============================================================
// 13. 编译期检查
// ============================================================

static_assert(Integer<int>);
static_assert(Integer<unsigned long>);
static_assert(!Integer<double>);

static_assert(Number<int>);
static_assert(Number<double>);
static_assert(!Number<std::string>);

static_assert(SignedInteger<int>);
static_assert(!SignedInteger<unsigned int>);

static_assert(Addable<int>);
static_assert(Addable<std::string>);
static_assert(Addable<Point>);
static_assert(!Addable<EmptyType>);

static_assert(StreamPrintable<int>);
static_assert(StreamPrintable<std::string>);
static_assert(StreamPrintable<User>);
static_assert(StreamPrintable<Point>);
static_assert(!StreamPrintable<EmptyType>);

static_assert(HasValueType<std::vector<int>>);
static_assert(HasValueType<std::list<std::string>>);
static_assert(!HasValueType<int>);

static_assert(ContainerLike<std::vector<int>>);
static_assert(ContainerLike<std::list<std::string>>);

// ============================================================
// 14. 主函数
// ============================================================

int main()
{
    std::cout << "===== 1. 四种约束语法 =====\n";

    std::cout << "twice_v1(10) = " << twice_v1(10) << '\n';
    std::cout << "twice_v2(10) = " << twice_v2(10) << '\n';
    std::cout << "twice_v3(10) = " << twice_v3(10) << '\n';
    std::cout << "twice_v4(10) = " << twice_v4(10) << '\n';
    std::cout << "square(6)   = " << square(6) << '\n';

    std::cout << "\n===== 2. 自定义 Concept =====\n";

    std::cout << "add(10, 20) = " << add(10, 20) << '\n';

    std::string hello = "Hello, ";
    std::string world = "Concepts";

    std::cout << "add(string, string) = "
        << add(hello, world)
        << '\n';

    Point point1{ 1, 2 };
    Point point2{ 3, 4 };
    Point point3 = add(point1, point2);

    printLine(point3);
    printLine(User{ "Alice" });

    std::cout << "\n===== 3. Concept 组合 =====\n";

    printNumber(42);
    printNumber(3.14);

    std::cout << "\n===== 4. 约束重载 =====\n";

    classify(std::string{ "hello" }); // generic type
    classify(3.14);                 // generic type
    classify(42U);                  // integral type
    classify(42);                   // signed integral type

    std::cout << "\n===== 5. 约束类模板 =====\n";

    Counter<int> counter;
    counter.add(10);
    counter.add(20);

    std::cout << "counter = " << counter.value() << '\n';

    std::cout << "\n===== 6. 约束部分成员 =====\n";

    Box<int> integerBox{ 100 };
    std::cout << "integerBox.add(23) = "
        << integerBox.add(23)
        << '\n';
    integerBox.print();

    Box<std::string> stringBox{ "Hello" };
    std::cout << "stringBox.add(...) = "
        << stringBox.add(", C++20")
        << '\n';
    stringBox.print();

    Box<User> userBox{ User{"Bob"} };

    // User 支持输出，因此可以调用 print()。
    userBox.print();

    // User 不支持 operator+，所以不能调用：
    // userBox.add(User{"Alice"});

    std::cout << "\n===== 7. Container / Range =====\n";

    const std::vector<int> numbers{ 1, 2, 3, 4, 5 };
    const std::list<std::string> words{
        "C++",
        "Concepts",
        "Ranges"
    };

    printContainer(numbers);
    printContainer(words);

    printRange(numbers);
    printRange(words);

    std::cout << "\n===== 8. invocable =====\n";

    const int result1 = execute(
        [](int a, int b)
        {
            return a + b;
        },
        10,
        20
    );

    std::cout << "execute result = " << result1 << '\n';

    const int result2 = executeIntegerFunction(
        [](int value)
        {
            return value * 10;
        },
        7
    );

    std::cout << "integer function result = "
        << result2
        << '\n';

    std::cout << "\n===== 9. noexcept Concept =====\n";

    std::cout << std::boolalpha;
    std::cout << "vector<int> has noexcept member swap: "
        << NoThrowMemberSwappable<std::vector<int>>
        << '\n';

    return 0;
}

总而言之，先concept $1 = requires(T a,T b){}

然后require $1或者写参数中

注意事项：

Concept 是编译期机制，通常没有运行时成本。
